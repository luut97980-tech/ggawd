cmake_minimum_required(VERSION 3.24)
project(VTMenuIOS LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_OBJCXX_STANDARD 20)
set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
option(VT_UI_TESTS_ONLY "Portable ImGui UI tests" OFF)
option(VT_BUILD_PLUGIN "Build a library loaded inside the host app" ON)
set(IMGUI ${CMAKE_CURRENT_SOURCE_DIR}/third_party/imgui)
set(IMGUI_CORE ${IMGUI}/imgui.cpp ${IMGUI}/imgui_draw.cpp ${IMGUI}/imgui_widgets.cpp ${IMGUI}/imgui_tables.cpp)
if(VT_UI_TESTS_ONLY)
  enable_testing()
  add_executable(mobile_ui_test tests/mobile_ui_test.cpp src/mobile_menu.cpp ${IMGUI_CORE})
  target_include_directories(mobile_ui_test PRIVATE src ${IMGUI})
  target_compile_definitions(mobile_ui_test PRIVATE IMGUI_ENABLE_TEST_ENGINE)
  if(MSVC)
    target_compile_options(mobile_ui_test PRIVATE /UNDEBUG /utf-8)
  else()
    target_compile_options(mobile_ui_test PRIVATE -UNDEBUG -Wall -Wextra)
  endif()
  add_test(NAME mobile_ui COMMAND mobile_ui_test)
  add_executable(backend_api_test tests/backend_api_test.cpp src/game_backend.cpp)
  target_include_directories(backend_api_test PRIVATE src ${IMGUI})
  if(MSVC)
    target_compile_options(backend_api_test PRIVATE /UNDEBUG /utf-8)
  else()
    target_compile_options(backend_api_test PRIVATE -UNDEBUG -Wall -Wextra)
  endif()
  add_test(NAME backend_api COMMAND backend_api_test)
  add_executable(app_metadata_test tests/app_metadata_test.cpp)
  target_include_directories(app_metadata_test PRIVATE src)
  if(MSVC)
    target_compile_options(app_metadata_test PRIVATE /UNDEBUG /utf-8)
  else()
    target_compile_options(app_metadata_test PRIVATE -UNDEBUG -Wall -Wextra)
  endif()
  add_test(NAME app_metadata COMMAND app_metadata_test)
  if(APPLE)
    enable_language(OBJCXX)
    add_executable(app_metadata_apple_test tests/app_metadata_apple_test.mm)
    target_include_directories(app_metadata_apple_test PRIVATE src)
    target_compile_options(app_metadata_apple_test PRIVATE -fobjc-arc -UNDEBUG)
    target_link_libraries(app_metadata_apple_test PRIVATE "-framework Foundation")
    add_test(NAME app_metadata_foundation COMMAND app_metadata_apple_test)
  endif()
  return()
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS" OR NOT CMAKE_GENERATOR STREQUAL "Xcode")
  message(FATAL_ERROR "Use macOS/Xcode with -DCMAKE_SYSTEM_NAME=iOS, or VT_UI_TESTS_ONLY=ON for portable tests.")
endif()
enable_language(OBJCXX)
set(VT_BUNDLE_ID "com.example.vtmenu.ios" CACHE STRING "Unique app bundle ID")
set(VT_DEVELOPMENT_TEAM "" CACHE STRING "Apple signing team ID")
if(VT_BUILD_PLUGIN)
  # Embed the Vietnamese font so only one dylib needs to be added to the IPA.
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/Resources/NotoSans-Regular.ttf" font_hex HEX)
  string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," font_values "${font_hex}")
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/generated_font.hpp"
    "#pragma once\nnamespace vt::ios {inline constexpr unsigned char FontBytes[]={${font_values}};}\n")
  add_library(VTMenu SHARED src/Plugin.mm src/game_backend.cpp src/mobile_menu.cpp ${IMGUI_CORE}
    ${IMGUI}/backends/imgui_impl_metal.mm)
  target_include_directories(VTMenu PRIVATE src ${IMGUI} ${IMGUI}/backends ${CMAKE_CURRENT_BINARY_DIR})
  target_compile_options(VTMenu PRIVATE $<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>)
  target_link_libraries(VTMenu PRIVATE "-framework UIKit" "-framework Foundation" "-framework CoreGraphics" "-framework Metal" "-framework MetalKit" "-framework QuartzCore")
  set_target_properties(VTMenu PROPERTIES PREFIX "" OUTPUT_NAME VTMenu
    MACOSX_RPATH TRUE INSTALL_NAME_DIR "@rpath" BUILD_WITH_INSTALL_NAME_DIR TRUE
    XCODE_ATTRIBUTE_IPHONEOS_DEPLOYMENT_TARGET "15.0"
    XCODE_ATTRIBUTE_CLANG_ENABLE_OBJC_ARC YES XCODE_ATTRIBUTE_ENABLE_BITCODE NO
    XCODE_ATTRIBUTE_CODE_SIGNING_ALLOWED NO XCODE_GENERATE_SCHEME TRUE)
  return()
endif()
message(FATAL_ERROR "This R5 package builds the in-process plugin. Use VT_BUILD_PLUGIN=ON.")
